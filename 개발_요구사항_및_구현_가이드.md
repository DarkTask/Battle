# 게임 시스템 개발 요구사항 및 구현 가이드

## 📋 요구사항 요약

### 핵심 변경사항
1. **1인 테스트 모드**: Host 혼자 A, B 플레이어 역할 모두 수행
2. **개선된 캐릭터 선택**: 3초 타이머, 자동 선택
3. **전투 순서 지정**: 비공개 슬롯 배치 (1번, 2번, 3번)
4. **다단계 전투 시스템**: 1v1 (3라운드) + 3v3 (최종)
5. **점수 시스템**: 1v1 승리 +1점, 3v3 승리 +2점
6. **순환 구조**: 결과 표시 → 캐릭터 선택으로 복귀

public class BattleGameManager : NetworkBehaviour
{
    [SyncVar] private float turnTimer = 3f;
    private const float TURN_TIME_LIMIT = 3f;
    
    void Update()
    {
        if (currentState == GameState.CharacterSelect)
        {
            if (isServer)
            {
                turnTimer -= Time.deltaTime;
                
                if (turnTimer <= 0)
                {
                    AutoSelectRandomChampion();
                    NextTurn();
                }
            }
            
            // UI 업데이트
            CharacterSelectUI.Instance.UpdateTimer(turnTimer);
        }
    }
    
    [Server]
    void AutoSelectRandomChampion()
    {
        // 선택 가능한 챔피언 중 랜덤 선택
        List<int> availableChampions = GetAvailableChampionIndices();
        int randomIndex = Random.Range(0, availableChampions.Count);
        SelectChampion(availableChampions[randomIndex]);
    }
}

```csharp
[Command(requiresAuthority = false)]
public void CmdSelectChampion(int championIndex, NetworkConnectionToClient sender = null)
{
    if (currentState != GameState.CharacterSelect) return;
    if (IsChampionAlreadySelected(championIndex)) return;
    
    // 현재 턴 플레이어 결정
    int currentPlayer = currentTurn % 2; // 0=A, 1=B
    
    // 챔피언 추가
    if (currentPlayer == 0)
        playerA.selectedChampions.Add(GetChampionData(championIndex));
    else
        playerB.selectedChampions.Add(GetChampionData(championIndex));
    
    // UI 업데이트
    RpcUpdateCharacterSelection(championIndex, currentPlayer);
    
    // 다음 턴
    NextTurn();
}

[Server]
void NextTurn()
{
    currentTurn++;
    turnTimer = TURN_TIME_LIMIT;
    
    if (currentTurn >= 6)
    {
        // 선택 완료 → 전투 순서 지정 단계로
        ChangeState(GameState.OrderSetup);
    }
    else
    {
        RpcUpdateTurnDisplay(currentTurn);
    }
}

[ClientRpc]
void RpcUpdateCharacterSelection(int championIndex, int playerIndex)
{
    CharacterSelectUI.Instance.DisableCharacter(championIndex);
    CharacterSelectUI.Instance.AddToPlayerSlot(championIndex, playerIndex);
}

public class DraggableChampion : MonoBehaviour, IBeginDragHandler, IDragHandler, IEndDragHandler
{
    public int championIndex;
    private Vector3 originalPosition;
    
    public void OnBeginDrag(PointerEventData eventData)
    {
        originalPosition = transform.position;
    }
    
    public void OnDrag(PointerEventData eventData)
    {
        transform.position = Input.mousePosition;
    }
    
    public void OnEndDrag(PointerEventData eventData)
    {
        // 슬롯에 드롭되었는지 확인
        OrderSlot slot = GetSlotUnderPointer();
        
        if (slot != null)
        {
            OrderSetupUI.Instance.AssignToSlot(championIndex, slot.slotIndex);
        }
        else
        {
            transform.position = originalPosition;
        }
    }
}

public class OrderSetupUI : MonoBehaviour
{
    private int[] playerOrder = new int[3] { -1, -1, -1 };
    
    public void AssignToSlot(int championIndex, int slotIndex)
    {
        // 이미 다른 슬롯에 있으면 제거
        for (int i = 0; i < 3; i++)
        {
            if (playerOrder[i] == championIndex)
                playerOrder[i] = -1;
        }
        
        playerOrder[slotIndex] = championIndex;
        UpdateVisuals();
    }
    
    public void ConfirmOrder()
    {
        if (IsOrderComplete())
        {
            BattleGameManager.Instance.CmdSetBattleOrder(playerOrder);
        }
    }
    
    bool IsOrderComplete()
    {
        return playerOrder[0] != -1 && playerOrder[1] != -1 && playerOrder[2] != -1;
    }
}
```

### Phase 4: 전투 시스템

#### 4.1 전투 시작

```csharp
[Server]
public void StartBattle(int round)
{
    currentRound = round;
    
    ChampionData champA, champB;
    
    if (round <= 3)
    {
        // 1v1 전투
        champA = playerA.GetChampionAtSlot(round - 1);
        champB = playerB.GetChampionAtSlot(round - 1);
        
        SpawnSingleBattle(champA, champB);
    }
    else
    {
        // 3v3 단체전
        SpawnTeamBattle(playerA.selectedChampions, playerB.selectedChampions);
    }
    
    RpcShowBattleUI(round);
}

[Server]
void SpawnSingleBattle(ChampionData champA, ChampionData champB)
{
    // 챔피언 스폰
    GameObject charA = Instantiate(champA.prefab, spawnPointA.position, Quaternion.identity);
    GameObject charB = Instantiate(champB.prefab, spawnPointB.position, Quaternion.identity);
    
    // 레이어 설정 (TeamA, TeamB)
    SetTeamLayer(charA, 24);
    SetTeamLayer(charB, 25);
    
    // 네트워크 스폰
    NetworkServer.Spawn(charA);
    NetworkServer.Spawn(charB);
    
    currentChampionA = charA;
    currentChampionB = charB;
    
    // 체력 이벤트 구독
    charA.GetComponent<Health>().OnDeath += () => OnChampionDeath(1); // B 승리
    charB.GetComponent<Health>().OnDeath += () => OnChampionDeath(0); // A 승리
}

[Server]
void SpawnTeamBattle(List<ChampionData> teamA, List<ChampionData> teamB)
{
    // 살아있는 챔피언만 스폰
    for (int i = 0; i < 3; i++)
    {
        if (teamA[i].isAlive)
        {
            Vector3 pos = spawnPointA.position + new Vector3(i * 2, 0, 0);
            GameObject charA = Instantiate(teamA[i].prefab, pos, Quaternion.identity);
            SetTeamLayer(charA, 24);
            NetworkServer.Spawn(charA);
            teamAChampions.Add(charA);
        }
        
        if (teamB[i].isAlive)
        {
            Vector3 pos = spawnPointB.position + new Vector3(i * 2, 0, 0);
            GameObject charB = Instantiate(teamB[i].prefab, pos, Quaternion.identity);
            SetTeamLayer(charB, 25);
            NetworkServer.Spawn(charB);
            teamBChampions.Add(charB);
        }
    }
    
    // 모든 챔피언에 죽음 이벤트 구독
    foreach (var champ in teamAChampions)
        champ.GetComponent<Health>().OnDeath += CheckTeamWipe;
    foreach (var champ in teamBChampions)
        champ.GetComponent<Health>().OnDeath += CheckTeamWipe;
}
```

#### 4.2 전투 종료 처리

```csharp
[Server]
void OnChampionDeath(int winnerId)
{
    // 점수 추가
    if (winnerId == 0)
        playerA.totalScore += 1;
    else
        playerB.totalScore += 1;
    
    // 챔피언 사망 표시
    int loserSlot = currentRound - 1;
    if (winnerId == 0)
        playerB.selectedChampions[playerB.battleOrder[loserSlot]].isAlive = false;
    else
        playerA.selectedChampions[playerA.battleOrder[loserSlot]].isAlive = false;
    
    // 정리
    CleanupBattle();
    
    // 다음 라운드
    StartCoroutine(NextRoundDelay());
}

[Server]
void CheckTeamWipe()
{
    int aliveTeamA = teamAChampions.Count(c => c.GetComponent<Health>().CurrentHealth > 0);
    int aliveTeamB = teamBChampions.Count(c => c.GetComponent<Health>().CurrentHealth > 0);
    
    if (aliveTeamA == 0)
    {
        // Team B 승리 (+2점)
        playerB.totalScore += 2;
        EndFinalBattle();
    }
    else if (aliveTeamB == 0)
    {
        // Team A 승리 (+2점)
        playerA.totalScore += 2;
        EndFinalBattle();
    }
}

[Server]
IEnumerator NextRoundDelay()
{
    yield return new WaitForSeconds(2f);
    
    currentRound++;
    
    if (currentRound <= 3)
    {
        ChangeState((GameState)(currentRound + 2)); // Battle_Round2 or Battle_Round3
        StartBattle(currentRound);
    }
    else
    {
        ChangeState(GameState.Battle_Final);
        StartBattle(4);
    }
}

[Server]
void EndFinalBattle()
{
    CleanupAllBattles();
    ChangeState(GameState.Result);
    ShowResult();
}
```
