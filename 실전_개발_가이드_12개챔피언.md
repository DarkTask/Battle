## 📋 조정된 사양

### 기본 설정
- **챔피언 수**: 12개 (기존 UI 그대로 사용)
- **선택 방식**: A → B → A → B → A → B (총 6턴)
- **각 플레이어**: 3개씩 선택
- **타이머**: 3초/턴 (자동 선택)

```
1. 캐릭터 선택 (6턴) → 2. 전투 순서 지정 → 3. 전투 (4라운드) → 4. 결과
```

```csharp
using UnityEngine;
using System.Collections.Generic;

[CreateAssetMenu(fileName = "ChampionDatabase", menuName = "Battle/Champion Database")]
public class ChampionDatabase : ScriptableObject
{
    public List<ChampionData> champions = new List<ChampionData>();
    
    public ChampionData GetChampion(int index)
    {
        if (index >= 0 && index < champions.Count)
            return champions[index];
        return null;
    }
}

[System.Serializable]
public class ChampionData
{
    public int id;
    public string championName;
    public Sprite icon;
    public Sprite characterImage;
    public GameObject championPrefab;
    
    [Header("Stats")]
    public int strength = 10;
    public int dexterity = 10;
    public int constitution = 10;
    
    [Header("Battle Stats")]
    public int maxHealth = 100;
    public int attackPower = 10;
    public float moveSpeed = 5f;
}
```

## 🎮 Step 2: BattleGameManager 구현 (30분)

**파일**: `Assets/Scripts/Battle/BattleGameManager.cs`

```csharp
using UnityEngine;
using Mirror;
using System.Collections.Generic;

public class BattleGameManager : NetworkBehaviour
{
    public static BattleGameManager Instance;
    
    [Header("Settings")]
    public bool soloTestMode = true;
    public ChampionDatabase championDB;
    
    [Header("Game State")]
    [SyncVar] public GameState currentState = GameState.CharacterSelect;
    [SyncVar] public int currentTurn = 0; // 0-5
    [SyncVar] public float turnTimer = 3f;
    
    [Header("Player Data")]
    public PlayerGameData playerA;
    public PlayerGameData playerB;
    
    // Constants
    private const float TURN_TIME_LIMIT = 3f;
    private const int TOTAL_TURNS = 6;
    
    void Awake()
    {
        if (Instance == null) Instance = this;
        else Destroy(gameObject);
        
        playerA = new PlayerGameData { playerIndex = 0 };
        playerB = new PlayerGameData { playerIndex = 1 };
    }
    
    void Start()
    {
        if (soloTestMode)
            Debug.Log("🎮 Solo Test Mode: 1인 테스트 모드 활성화");
        
        if (isServer)
            StartCharacterSelect();
    }
    
    void Update()
    {
        if (!isServer) return;
        
        if (currentState == GameState.CharacterSelect)
        {
            UpdateCharacterSelectTimer();
        }
    }
    
    [Server]
    void StartCharacterSelect()
    {
        currentState = GameState.CharacterSelect;
        currentTurn = 0;
        turnTimer = TURN_TIME_LIMIT;
        
        RpcUpdateUI();
    }
    
    [Server]
    void UpdateCharacterSelectTimer()
    {
        turnTimer -= Time.deltaTime;
        
        if (turnTimer <= 0)
        {
            Debug.Log($"⏰ 시간 초과! 자동 선택");
            AutoSelectChampion();
            NextTurn();
        }
    }
    
    [Command(requiresAuthority = false)]
    public void CmdSelectChampion(int championIndex, NetworkConnectionToClient sender = null)
    {
        if (currentState != GameState.CharacterSelect) return;
        if (IsChampionAlreadySelected(championIndex)) return;
        
        // 현재 턴의 플레이어
        int currentPlayer = currentTurn % 2; // 0=A, 1=B
        
        // 챔피언 추가
        ChampionData champion = championDB.GetChampion(championIndex);
        if (currentPlayer == 0)
            playerA.selectedChampions.Add(champion);
        else
            playerB.selectedChampions.Add(champion);
        
        Debug.Log($"✅ Player {(currentPlayer == 0 ? "A" : "B")} 선택: {champion.championName}");
        
        // UI 업데이트
        RpcUpdateChampionSelection(championIndex, currentPlayer);
        
        // 다음 턴
        NextTurn();
    }
    
    [Server]
    void AutoSelectChampion()
    {
        List<int> available = GetAvailableChampionIndices();
        if (available.Count > 0)
        {
            int randomIndex = available[Random.Range(0, available.Count)];
            
            // 자동 선택 실행
            int currentPlayer = currentTurn % 2;
            ChampionData champion = championDB.GetChampion(randomIndex);
            
            if (currentPlayer == 0)
                playerA.selectedChampions.Add(champion);
            else
                playerB.selectedChampions.Add(champion);
            
            Debug.Log($"🤖 자동 선택: {champion.championName}");
            RpcUpdateChampionSelection(randomIndex, currentPlayer);
        }
    }
    
    [Server]
    void NextTurn()
    {
        currentTurn++;
        turnTimer = TURN_TIME_LIMIT;
        
        if (currentTurn >= TOTAL_TURNS)
        {
            // 선택 완료 → 전투 순서 지정으로
            Debug.Log("✅ 캐릭터 선택 완료!");
            ChangeState(GameState.OrderSetup);
        }
        else
        {
            RpcUpdateUI();
        }
    }
    
    bool IsChampionAlreadySelected(int index)
    {
        foreach (var champ in playerA.selectedChampions)
            if (champ.id == index) return true;
        foreach (var champ in playerB.selectedChampions)
            if (champ.id == index) return true;
        return false;
    }
    
    List<int> GetAvailableChampionIndices()
    {
        List<int> available = new List<int>();
        for (int i = 0; i < 12; i++)
        {
            if (!IsChampionAlreadySelected(i))
                available.Add(i);
        }
        return available;
    }
    
    [Server]
    void ChangeState(GameState newState)
    {
        currentState = newState;
        Debug.Log($"🎮 State Changed: {newState}");
        RpcOnStateChanged(newState);
    }
    
    [ClientRpc]
    void RpcUpdateUI()
    {
        // UI 매니저에 업데이트 알림
        CharacterSelectUI.Instance?.UpdateTurnDisplay(currentTurn);
        CharacterSelectUI.Instance?.UpdateTimer(turnTimer);
    }
    
    [ClientRpc]
    void RpcUpdateChampionSelection(int championIndex, int playerIndex)
    {
        CharacterSelectUI.Instance?.OnChampionSelected(championIndex, playerIndex);
    }
    
    [ClientRpc]
    void RpcOnStateChanged(GameState newState)
    {
        Debug.Log($"[Client] State: {newState}");
        
        switch (newState)
        {
            case GameState.OrderSetup:
                // TODO: OrderSetupPanel 표시
                break;
            case GameState.Battle_Round1:
                // TODO: 전투 시작
                break;
        }
    }
}

public enum GameState
{
    Lobby,
    CharacterSelect,
    OrderSetup,
    Battle_Round1,
    Battle_Round2,
    Battle_Round3,
    Battle_Final,
    Result
}

[System.Serializable]
public class PlayerGameData
{
    public int playerIndex;
    public List<ChampionData> selectedChampions = new List<ChampionData>();
    public int[] battleOrder = new int[3];
    public int totalScore;
}
```

```csharp
using UnityEngine;
using UnityEngine.UI;
using TMPro;

public class CharacterElement : MonoBehaviour
{
    [Header("References")]
    public int championIndex;
    public Image cardBackground;      // Card_Red/Blue/Orange
    public Image championIcon;        // Icon
    public Image characterImage;      // Character_Mask > Character
    public GameObject starOn;         // Star_On
    public GameObject starOff;        // Star_Off
    public TextMeshProUGUI nameText;  // Text_Name
    public GameObject backGlow;       // BackGlow
    
    [Header("Stats")]
    public TextMeshProUGUI strText;
    public TextMeshProUGUI dexText;
    public TextMeshProUGUI conText;
    
    [Header("Sprites")]
    public Sprite cardOrange;
    public Sprite cardRed;
    public Sprite cardBlue;
    
    private Button button;
    private bool isSelected = false;
    
    void Awake()
    {
        button = GetComponent<Button>();
        if (button == null)
            button = gameObject.AddComponent<Button>();
        
        button.onClick.AddListener(OnClicked);
    }
    
    public void Initialize(ChampionData data, int index)
    {
        championIndex = index;
        
        nameText.text = data.championName;
        championIcon.sprite = data.icon;
        if (characterImage != null && data.characterImage != null)
            characterImage.sprite = data.characterImage;
        
        // 스탯 표시
        if (strText != null) strText.text = $"STR:{data.strength}";
        if (dexText != null) dexText.text = $"DEX:{data.dexterity}";
        if (conText != null) conText.text = $"CON:{data.constitution}";
        
        SetSelected(false, -1);
    }
    
    void OnClicked()
    {
        if (isSelected)
        {
            Debug.Log($"이미 선택된 챔피언: {nameText.text}");
            return;
        }
        
        Debug.Log($"챔피언 선택 시도: {nameText.text} (Index: {championIndex})");
        
        // BattleGameManager에 선택 알림
        if (BattleGameManager.Instance != null)
        {
            BattleGameManager.Instance.CmdSelectChampion(championIndex);
        }
    }
    
    public void SetSelected(bool selected, int playerIndex)
    {
        isSelected = selected;
        
        // Star 표시
        if (starOn != null) starOn.SetActive(selected);
        if (starOff != null) starOff.SetActive(!selected);
        
        // 카드 색상 변경
        if (selected)
        {
            if (playerIndex == 0 && cardRed != null)
                cardBackground.sprite = cardRed;
            else if (playerIndex == 1 && cardBlue != null)
                cardBackground.sprite = cardBlue;
            
            // Glow 효과
            if (backGlow != null)
                backGlow.SetActive(true);
        }
        else
        {
            if (cardOrange != null)
                cardBackground.sprite = cardOrange;
            
            if (backGlow != null)
                backGlow.SetActive(false);
        }
        
        // 버튼 비활성화
        button.interactable = !selected;
    }
}
```

### 3-2. CharacterSelectUI Manager

**파일**: `Assets/Scripts/Battle/UI/CharacterSelectUI.cs`

```csharp
using UnityEngine;
using TMPro;

public class CharacterSelectUI : MonoBehaviour
{
    public static CharacterSelectUI Instance;
    
    [Header("References")]
    public ChampionDatabase championDB;
    public CharacterElement[] characterElements = new CharacterElement[12];
    
    [Header("UI Elements")]
    public TextMeshProUGUI gameText;           // "Player A의 차례"
    public TextMeshProUGUI timerText;          // "03초"
    
    [Header("Side Panels")]
    public Transform playerAPanel;             // Panel_Left
    public Transform playerBPanel;             // Panel_Right
    public GameObject[] playerASlots = new GameObject[3];
    public GameObject[] playerBSlots = new GameObject[3];
    
    void Awake()
    {
        if (Instance == null) Instance = this;
    }
    
    void Start()
    {
        InitializeChampions();
    }
    
    void InitializeChampions()
    {
        for (int i = 0; i < characterElements.Length && i < 12; i++)
        {
            ChampionData data = championDB.GetChampion(i);
            if (data != null)
            {
                characterElements[i].Initialize(data, i);
            }
        }
        
        Debug.Log("✅ CharacterSelectUI 초기화 완료");
    }
    
    public void UpdateTurnDisplay(int turn)
    {
        int currentPlayer = turn % 2;
        string playerName = currentPlayer == 0 ? "Player A" : "Player B";
        
        if (gameText != null)
        {
            gameText.text = $"{playerName}의 차례";
            gameText.color = currentPlayer == 0 ? Color.red : Color.blue;
        }
        
        Debug.Log($"🎯 Turn {turn}: {playerName}");
    }
    
    public void UpdateTimer(float time)
    {
        if (timerText != null)
        {
            int seconds = Mathf.CeilToInt(time);
            timerText.text = $"{seconds:D2}초";
            
            // 1초 이하면 빨간색
            if (time <= 1f)
                timerText.color = Color.red;
            else
                timerText.color = Color.white;
        }
    }
    
    public void OnChampionSelected(int championIndex, int playerIndex)
    {
        // 카드 UI 업데이트
        if (championIndex >= 0 && championIndex < characterElements.Length)
        {
            characterElements[championIndex].SetSelected(true, playerIndex);
        }
        
        // Side Panel 업데이트
        UpdateSidePanel(playerIndex);
    }
    
    void UpdateSidePanel(int playerIndex)
    {
        PlayerGameData playerData = playerIndex == 0 ? 
            BattleGameManager.Instance.playerA : 
            BattleGameManager.Instance.playerB;
        
        GameObject[] slots = playerIndex == 0 ? playerASlots : playerBSlots;
        
        for (int i = 0; i < slots.Length && i < playerData.selectedChampions.Count; i++)
        {
            ChampionData champion = playerData.selectedChampions[i];
            
            // 슬롯에 아이콘 표시
            Image iconImage = slots[i].GetComponentInChildren<Image>();
            if (iconImage != null && champion.icon != null)
            {
                iconImage.sprite = champion.icon;
                iconImage.enabled = true;
            }
            
            // 이름 표시 (있다면)
            TextMeshProUGUI nameText = slots[i].GetComponentInChildren<TextMeshProUGUI>();
            if (nameText != null)
            {
                nameText.text = champion.championName;
            }
        }
    }
}
```

## 🐛 트러블슈팅

### 문제 1: 카드 클릭이 안 됨
**해결**:
- Canvas에 GraphicRaycaster 있는지 확인
- EventSystem이 씬에 있는지 확인
- Button 컴포넌트 활성화 확인

### 문제 2: 네트워크 오류
**해결**:
- BattleGameManager에 NetworkIdentity 추가
- 씬에 NetworkManager (MatchNetworkManager) 있는지 확인

### 문제 3: 타이머가 업데이트 안 됨
**해결**:
- CharacterSelectUI.Instance가 null인지 확인
- Update()가 isServer에서만 실행되는지 확인

### 문제 4: UI 참조가 None
**해결**:
- Inspector에서 모든 참조 다시 드래그
- Prefab Mode에서 수정했는지 확인
